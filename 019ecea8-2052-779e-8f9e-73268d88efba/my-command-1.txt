Source Excel formula for your calculation:

k=IF(IF(p&lt;1,1,ROUNDUP(p,0))&lt;3,0,ROUNDUP(LOG(IF(p&lt;1,1,ROUNDUP(p,0))*4/3,2),0)-1-1)

c=IF(d&lt;2,2,ROUNDUP(d,0))-1

Rmin=MIN(IF(MOD(IF(n&lt;4,4,ROUNDUP(n,0)),2)=1,IF(n&lt;4,4,ROUNDUP(n,0)),IF(n&lt;4,4,ROUNDUP(n,0))-1),ROUNDUP(LOG(IF(n&lt;4,4,ROUNDUP(n,0))*4/3,2),0)+k)

Rmax=MIN(IF(MOD(IF(n&lt;4,4,ROUNDUP(n,0)),2)=1,IF(n&lt;4,4,ROUNDUP(n,0)),IF(n&lt;4,4,ROUNDUP(n,0))-1),ROUNDUP(LOG(IF(n&lt;4,4,ROUNDUP(n,0))*4/3,2),0)+(ROUNDUP(LOG(IF(n&lt;4,4,ROUNDUP(n,0))*4/3,2),0)-1-1),Rmin+c)
